<task type="checkpoint:decision" gate="blocking">
  <decision>Email capture backend — where to store waitlist signups</decision>
  <context>Need a free/cheap solution that captures emails reliably and allows sending nurture sequences later.</context>
  <options>
    <option id="resend-audiences">
      <name>Resend Audiences (free tier: 100 contacts, then $0.20/contact)</name>
      <pros>Simple API, great DX, built-in audience management, can send sequences later. Modern developer tool.</pros>
      <cons>100 free contact limit is tight — need to upgrade early. Newer product.</cons>
    </option>
    <option id="loops">
      <name>Loops.so (free tier: 1,000 contacts)</name>
      <pros>1,000 free contacts, built-in email sequences, good landing page integrations, designed for waitlists.</pros>
      <cons>Less developer-focused API, another SaaS account to manage.</cons>
    </option>
    <option id="supabase">
      <name>Supabase table + Resend for emails</name>
      <pros>Full control, free tier generous (50K rows), can build custom dashboard. Own your data.</pros>
      <cons>More setup, need to build email sending separately, no built-in sequence management.</cons>
    </option>
  </options>
  <resume-signal>Select: resend-audiences, loops, or supabase</resume-signal>
</task>
